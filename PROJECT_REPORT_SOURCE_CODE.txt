import { ErrorExplanation, LiveDiagnostic, SupportedLanguage } from './types';

/**
 * Real-Time As-You-Type Code Analyzer & Linter for Multi-Language Sandboxes
 * Supports: Python, Java, C++, C, Go, Rust, JavaScript, TypeScript
 * Scans code while typing and immediately reports:
 * - Has error? (true/false)
 * - Where the mistake was made (exact line & snippet)
 * - Why it is an error in this specific language
 * - How to fix it
 */
export function analyzeCodeLive(
  code: string,
  language: SupportedLanguage = 'javascript',
  challengeId?: string
): LiveDiagnostic {
  if (!code || !code.trim()) {
    return {
      hasError: false,
      type: 'clean',
      title: 'Empty Editor',
      whereIsMistake: 'Editor is empty.',
      whyIsError: 'No code to evaluate yet.',
      howToFix: 'Start typing code or click "Load Starter" to load template code.',
    };
  }

  const lines = code.split('\n');

  // ==========================================
  // 1. PYTHON MODE (sandbox.py)
  // ==========================================
  if (language === 'python') {
    let openParens = 0;
    let openBrackets = 0;
    let openBraces = 0;

    for (let i = 0; i < lines.length; i++) {
      const lineText = lines[i];
      const lineTrimmed = lineText.trim();
      const lineNum = i + 1;

      // Check for JavaScript/Java console.log in Python
      if (lineTrimmed.includes('console.log(')) {
        return {
          hasError: true,
          type: 'syntax',
          title: "NameError: 'console' is not defined in Python",
          line: lineNum,
          codeSnippet: lineTrimmed,
          whereIsMistake: `Line ${lineNum}: "${lineTrimmed}"`,
          whyIsError: "In Python, output is printed using the 'print(...)' function. 'console.log' is JavaScript syntax.",
          howToFix: `Replace 'console.log(...)' with 'print(...)'.`,
        };
      }

      // Check for Java System.out.println in Python
      if (lineTrimmed.includes('System.out.println(')) {
        return {
          hasError: true,
          type: 'syntax',
          title: "NameError: 'System' is not defined in Python",
          line: lineNum,
          codeSnippet: lineTrimmed,
          whereIsMistake: `Line ${lineNum}: "${lineTrimmed}"`,
          whyIsError: "In Python, output is printed using 'print(...)'. 'System.out.println' is Java syntax.",
          howToFix: `Replace 'System.out.println(...)' with 'print(...)'.`,
        };
      }

      // Check for JS/Java variable keywords
      if (lineTrimmed.startsWith('let ') || lineTrimmed.startsWith('const ') || lineTrimmed.startsWith('var ')) {
        return {
          hasError: true,
          type: 'syntax',
          title: "SyntaxError: 'let'/'const'/'var' is not valid in Python",
          line: lineNum,
          codeSnippet: lineTrimmed,
          whereIsMistake: `Line ${lineNum}: "${lineTrimmed}"`,
          whyIsError: "Python does not use variable declaration keywords like 'let', 'const', or 'var'. Variables are created by direct assignment.",
          howToFix: `Remove the keyword: '${lineTrimmed.replace(/^(let|const|var)\s+/, '')}'.`,
        };
      }

      // Check for JS/Java comments '//'
      if (lineTrimmed.startsWith('//')) {
        return {
          hasError: true,
          type: 'syntax',
          title: "SyntaxError: '//' is floor division in Python, not a comment",
          line: lineNum,
          codeSnippet: lineTrimmed,
          whereIsMistake: `Line ${lineNum}: "${lineTrimmed}"`,
          whyIsError: "In Python, single-line comments begin with the '#' symbol. '//' is the floor-division operator.",
          howToFix: `Change '//' to '#': "# ${lineTrimmed.replace(/^\/\/\s*/, '')}".`,
        };
      }

      // Check for Java import packages
      if (lineTrimmed.startsWith('import java.') || lineTrimmed.includes('import .')) {
        return {
          hasError: true,
          type: 'syntax',
          title: "ModuleNotFoundError: Java package in Python",
          line: lineNum,
          codeSnippet: lineTrimmed,
          whereIsMistake: `Line ${lineNum}: "${lineTrimmed}"`,
          whyIsError: "Python does not have Java packages. Standard Python modules include 'math', 'json', 'sys', 'os', 'datetime'.",
          howToFix: "Use Python imports: 'import math' or 'from datetime import datetime'.",
        };
      }

      // Check for curly braces on print: print{
      if (lineTrimmed.startsWith('print{') || lineTrimmed.startsWith('print {')) {
        return {
          hasError: true,
          type: 'syntax',
          title: "SyntaxError: invalid syntax '{'",
          line: lineNum,
          codeSnippet: lineTrimmed,
          whereIsMistake: `Line ${lineNum}: "${lineTrimmed}"`,
          whyIsError: "Function calls in Python use round parentheses '()', not curly braces '{}'. Curly braces are reserved for dicts and sets.",
          howToFix: `Change to round parentheses: 'print(...)'.`,
        };
      }

      // Check for missing colon on def/if/for/while/class/elif/else/try/except
      if (
        (lineTrimmed.startsWith('def ') ||
          lineTrimmed.startsWith('if ') ||
          lineTrimmed.startsWith('elif ') ||
          lineTrimmed.startsWith('for ') ||
          lineTrimmed.startsWith('while ') ||
          lineTrimmed === 'else' ||
          lineTrimmed === 'try' ||
          lineTrimmed.startsWith('except')) &&
        !lineTrimmed.endsWith(':') &&
        !lineTrimmed.endsWith('"""') &&
        !lineTrimmed.endsWith("'''")
      ) {
        return {
          hasError: true,
          type: 'syntax',
          title: "SyntaxError: expected ':' at end of compound statement",
          line: lineNum,
          codeSnippet: lineTrimmed,
          whereIsMistake: `Line ${lineNum}: "${lineTrimmed}" (missing ':' at the end)`,
          whyIsError: "In Python, block headers (if, for, while, def, class, else, except) must always end with a colon ':'.",
          howToFix: `Add ':' at the end of the line: "${lineTrimmed}:".`,
        };
      }

      // Count delimiters
      for (const char of lineText) {
        if (char === '(') openParens++;
        if (char === ')') openParens--;
        if (char === '[') openBrackets++;
        if (char === ']') openBrackets--;
        if (char === '{') openBraces++;
        if (char === '}') openBraces--;
      }

      // If a print statement is unclosed on this line
      if (lineTrimmed.startsWith('print(') && !lineTrimmed.includes(')')) {
        return {
          hasError: true,
          type: 'syntax',
          title: "SyntaxError: unexpected EOF while parsing - unclosed '('",
          line: lineNum,
          codeSnippet: lineTrimmed,
          whereIsMistake: `Line ${lineNum}: "${lineTrimmed}"`,
          whyIsError: "The 'print(' statement was opened but never closed with a matching ')'. Python requires all opened parentheses to be closed before statement execution.",
          howToFix: `Close the parenthesis: "${lineTrimmed})".`,
        };
      }
    }

    if (openParens > 0) {
      return {
        hasError: true,
        type: 'syntax',
        title: "SyntaxError: unexpected EOF while parsing (unclosed parenthesis ')')",
        line: lines.length,
        codeSnippet: lines[lines.length - 1].trim(),
        whereIsMistake: `Line ${lines.length}: Unclosed parenthesis '('`,
        whyIsError: "There is an opening '(' in your Python code that has not been closed with a matching ')'.",
        howToFix: "Add ')' to close the open expression.",
      };
    }
  }

  // ==========================================
  // 2. JAVA MODE (Main.java)
  // ==========================================
  if (language === 'java') {
    let hasClass = false;

    for (let i = 0; i < lines.length; i++) {
      const lineText = lines[i];
      const lineTrimmed = lineText.trim();
      const lineNum = i + 1;

      if (lineTrimmed.includes('class ')) hasClass = true;

      // Check for Python print in Java
      if (lineTrimmed.startsWith('print(') || lineTrimmed.startsWith('console.log(')) {
        return {
          hasError: true,
          type: 'syntax',
          title: "error: cannot find symbol method print()",
          line: lineNum,
          codeSnippet: lineTrimmed,
          whereIsMistake: `Line ${lineNum}: "${lineTrimmed}"`,
          whyIsError: "Java does not have a bare 'print()' or 'console.log()' function. Standard output uses System.out.println(...).",
          howToFix: `Change to Java syntax: "System.out.println(${lineTrimmed.replace(/^(print|console\.log)\(/, '')}".`,
        };
      }

      // Check for Python def in Java
      if (lineTrimmed.startsWith('def ')) {
        return {
          hasError: true,
          type: 'syntax',
          title: "error: invalid method declaration in Java",
          line: lineNum,
          codeSnippet: lineTrimmed,
          whereIsMistake: `Line ${lineNum}: "${lineTrimmed}"`,
          whyIsError: "Java is statically typed and does not use Python 'def'. Java methods must specify a return type (e.g. void, int, String).",
          howToFix: "Declare as: public static void methodName() { ... }",
        };
      }

      // Check for malformed import
      if (lineTrimmed.includes('import .') || (lineTrimmed.startsWith('import ') && !lineTrimmed.endsWith(';') && !lineTrimmed.startsWith('import java.'))) {
        return {
          hasError: true,
          type: 'syntax',
          title: "error: '.' unexpected after import in Java",
          line: lineNum,
          codeSnippet: lineTrimmed,
          whereIsMistake: `Line ${lineNum}: "${lineTrimmed}"`,
          whyIsError: "Java package imports cannot have a dot immediately after 'import'. Java imports follow: 'import java.util.*;' or 'import java.util.Scanner;'.",
          howToFix: "Use correct Java import syntax: 'import java.util.*;'.",
        };
      }

      // Check for missing semicolon on plain statement lines
      if (
        lineTrimmed.length > 0 &&
        !lineTrimmed.startsWith('//') &&
        !lineTrimmed.startsWith('/*') &&
        !lineTrimmed.startsWith('*') &&
        !lineTrimmed.endsWith('{') &&
        !lineTrimmed.endsWith('}') &&
        !lineTrimmed.endsWith(';') &&
        !lineTrimmed.startsWith('public class') &&
        !lineTrimmed.startsWith('class ') &&
        !lineTrimmed.startsWith('static class') &&
        !lineTrimmed.startsWith('public static void main') &&
        !lineTrimmed.startsWith('public ') &&
        !lineTrimmed.startsWith('private ') &&
        !lineTrimmed.startsWith('protected ') &&
        !lineTrimmed.startsWith('import ') &&
        !lineTrimmed.startsWith('package ') &&
        !lineTrimmed.startsWith('@')
      ) {
        return {
          hasError: true,
          type: 'syntax',
          title: "error: ';' expected",
          line: lineNum,
          codeSnippet: lineTrimmed,
          whereIsMistake: `Line ${lineNum}: "${lineTrimmed}"`,
          whyIsError: "In Java, every statement must terminate with a semicolon ';'.",
          howToFix: `Add ';' to the end of line ${lineNum}: "${lineTrimmed};".`,
        };
      }
    }

    if (!hasClass && lines.some((l) => l.trim().length > 0)) {
      return {
        hasError: true,
        type: 'syntax',
        title: "error: class, interface, or enum expected",
        line: 1,
        codeSnippet: lines[0].trim(),
        whereIsMistake: "Line 1: Code written outside of a Java class",
        whyIsError: "In Java, all executable statements and methods must reside inside an enclosing class.",
        howToFix: "Enclose your code inside a class: public class Main { public static void main(String[] args) { ... } }.",
      };
    }
  }

  // ==========================================
  // 3. C++ MODE (main.cpp)
  // ==========================================
  if (language === 'cpp') {
    for (let i = 0; i < lines.length; i++) {
      const lineText = lines[i];
      const lineTrimmed = lineText.trim();
      const lineNum = i + 1;

      // Check for Java/JS output in C++
      if (lineTrimmed.includes('System.out.println(') || lineTrimmed.includes('console.log(')) {
        return {
          hasError: true,
          type: 'syntax',
          title: "error: unknown output identifier in C++",
          line: lineNum,
          codeSnippet: lineTrimmed,
          whereIsMistake: `Line ${lineNum}: "${lineTrimmed}"`,
          whyIsError: "In C++, output is streamed via 'std::cout << ... << std::endl;'. Neither System.out.println nor console.log exist in C++.",
          howToFix: 'Use C++ iostream: std::cout << "..." << std::endl;',
        };
      }

      // Check for missing semicolon on statements
      if (
        lineTrimmed.length > 0 &&
        !lineTrimmed.startsWith('//') &&
        !lineTrimmed.startsWith('#') &&
        !lineTrimmed.endsWith('{') &&
        !lineTrimmed.endsWith('}') &&
        !lineTrimmed.endsWith(';') &&
        !lineTrimmed.includes('int main(') &&
        !lineTrimmed.includes('float ') &&
        !lineTrimmed.includes('void ')
      ) {
        return {
          hasError: true,
          type: 'syntax',
          title: "error: expected ';' before end of line",
          line: lineNum,
          codeSnippet: lineTrimmed,
          whereIsMistake: `Line ${lineNum}: "${lineTrimmed}"`,
          whyIsError: "In C++, statements must terminate with a semicolon ';'.",
          howToFix: `Append ';' to line ${lineNum}: "${lineTrimmed};".`,
        };
      }
    }
  }

  // ==========================================
  // 4. C MODE (main.c)
  // ==========================================
  if (language === 'c') {
    for (let i = 0; i < lines.length; i++) {
      const lineTrimmed = lines[i].trim();
      const lineNum = i + 1;

      if (lineTrimmed.startsWith('print(') || lineTrimmed.startsWith('cout <<')) {
        return {
          hasError: true,
          type: 'syntax',
          title: "error: 'print' undeclared in C",
          line: lineNum,
          codeSnippet: lineTrimmed,
          whereIsMistake: `Line ${lineNum}: "${lineTrimmed}"`,
          whyIsError: "In C, standard formatted output is performed with 'printf(...)'. Include <stdio.h> to use printf.",
          howToFix: `Use printf: printf("Hello, World!\\n");`,
        };
      }

      if (
        lineTrimmed.length > 0 &&
        !lineTrimmed.startsWith('//') &&
        !lineTrimmed.startsWith('#') &&
        !lineTrimmed.endsWith('{') &&
        !lineTrimmed.endsWith('}') &&
        !lineTrimmed.endsWith(';') &&
        !lineTrimmed.includes('int main(')
      ) {
        return {
          hasError: true,
          type: 'syntax',
          title: "error: expected ';' before end of line in C",
          line: lineNum,
          codeSnippet: lineTrimmed,
          whereIsMistake: `Line ${lineNum}: "${lineTrimmed}"`,
          whyIsError: "In C, every statement must end with a semicolon ';'.",
          howToFix: `Add ';' to line ${lineNum}: "${lineTrimmed};".`,
        };
      }
    }
  }

  // ==========================================
  // 5. GO MODE (main.go)
  // ==========================================
  if (language === 'go') {
    let hasPackage = false;
    for (let i = 0; i < lines.length; i++) {
      const lineTrimmed = lines[i].trim();
      const lineNum = i + 1;
      if (lineTrimmed.startsWith('package ')) hasPackage = true;

      if (lineTrimmed.startsWith('print(') || lineTrimmed.startsWith('console.log(')) {
        return {
          hasError: true,
          type: 'syntax',
          title: "undefined: print (use fmt.Println in Go)",
          line: lineNum,
          codeSnippet: lineTrimmed,
          whereIsMistake: `Line ${lineNum}: "${lineTrimmed}"`,
          whyIsError: "In Go, formatted output uses the 'fmt' package: fmt.Println(...).",
          howToFix: `Use: fmt.Println(...) and import "fmt".`,
        };
      }
    }

    if (!hasPackage && lines.some((l) => l.trim().length > 0)) {
      return {
        hasError: true,
        type: 'syntax',
        title: "expected 'package', found import or statement",
        line: 1,
        codeSnippet: lines[0].trim(),
        whereIsMistake: "Line 1: Missing package clause",
        whyIsError: "Every Go source file must start with a package declaration: 'package main'.",
        howToFix: "Add 'package main' at line 1.",
      };
    }
  }

  // ==========================================
  // 6. RUST MODE (main.rs)
  // ==========================================
  if (language === 'rust') {
    for (let i = 0; i < lines.length; i++) {
      const lineTrimmed = lines[i].trim();
      const lineNum = i + 1;

      if (lineTrimmed.startsWith('print(') || lineTrimmed.startsWith('console.log(')) {
        return {
          hasError: true,
          type: 'syntax',
          title: "cannot find function `print` in this scope (use println! macro)",
          line: lineNum,
          codeSnippet: lineTrimmed,
          whereIsMistake: `Line ${lineNum}: "${lineTrimmed}"`,
          whyIsError: "In Rust, printing to console uses the macro 'println!(...)'. Note the exclamation mark '!' indicating a macro.",
          howToFix: `Use: println!("Hello from Rust!");`,
        };
      }
    }
  }

  // ==========================================
  // 7. JAVASCRIPT & TYPESCRIPT MODE (sandbox.js / sandbox.ts)
  // ==========================================
  if (language === 'javascript' || language === 'typescript') {
    let openParens = 0;
    let openBrackets = 0;
    let openBraces = 0;

    for (let i = 0; i < lines.length; i++) {
      const lineText = lines[i];
      const lineTrimmed = lineText.trim();
      const lineNum = i + 1;

      for (const ch of lineText) {
        if (ch === '(') openParens++;
        if (ch === ')') openParens--;
        if (ch === '[') openBrackets++;
        if (ch === ']') openBrackets--;
        if (ch === '{') openBraces++;
        if (ch === '}') openBraces--;
      }

      // Check for unclosed print( in JavaScript
      if (lineTrimmed.startsWith('print(') && !lineTrimmed.includes(')')) {
        return {
          hasError: true,
          type: 'syntax',
          title: "SyntaxError: missing ')' after argument list",
          line: lineNum,
          codeSnippet: lineTrimmed,
          whereIsMistake: `Line ${lineNum}: "${lineTrimmed}" (unclosed parenthesis)`,
          whyIsError: "You called 'print(' with an unclosed parenthesis. In addition, note that JavaScript uses 'console.log(...)' for output, while 'print(...)' is Python. If you want to write Python, click the 🐍 Python button in the language toolbar!",
          howToFix: `Close with ')': "console.log(...)" or switch language toolbar to Python.`,
        };
      }

      // Check for Python def in JavaScript
      if (lineTrimmed.startsWith('def ') && lineTrimmed.endsWith(':')) {
        return {
          hasError: true,
          type: 'syntax',
          title: "SyntaxError: Python 'def' function syntax in JavaScript",
          line: lineNum,
          codeSnippet: lineTrimmed,
          whereIsMistake: `Line ${lineNum}: "${lineTrimmed}"`,
          whyIsError: "You are using Python 'def func():' syntax in a JavaScript environment. JavaScript uses 'function name() { ... }'.",
          howToFix: "Change to: function name() { ... } or switch the language toolbar to Python.",
        };
      }

      // Check for Java import in JavaScript
      if (lineTrimmed.includes('import .') || lineTrimmed.startsWith('import java.')) {
        return {
          hasError: true,
          type: 'syntax',
          title: "SyntaxError: Java package import in JavaScript",
          line: lineNum,
          codeSnippet: lineTrimmed,
          whereIsMistake: `Line ${lineNum}: "${lineTrimmed}"`,
          whyIsError: "You wrote Java package syntax ('import . java'). In JavaScript, standard library utilities (Math, JSON, Date, Array) are built-in globally without imports.",
          howToFix: "Remove the Java import statement. Switch to Java in the language toolbar if writing Java.",
        };
      }

      // Check for Java System.out in JavaScript
      if (lineTrimmed.includes('System.out.println')) {
        return {
          hasError: true,
          type: 'syntax',
          title: "ReferenceError: System is not defined in JavaScript",
          line: lineNum,
          codeSnippet: lineTrimmed,
          whereIsMistake: `Line ${lineNum}: "${lineTrimmed}"`,
          whyIsError: "'System.out.println' is Java syntax. JavaScript uses 'console.log(...)'.",
          howToFix: "Replace 'System.out.println(...)' with 'console.log(...)'.",
        };
      }

      // Check for C++ #include in JavaScript
      if (lineTrimmed.startsWith('#include')) {
        return {
          hasError: true,
          type: 'syntax',
          title: "SyntaxError: C++ preprocessor directive in JavaScript",
          line: lineNum,
          codeSnippet: lineTrimmed,
          whereIsMistake: `Line ${lineNum}: "${lineTrimmed}"`,
          whyIsError: "'#include' is C++ syntax. JavaScript does not have preprocessor directives.",
          howToFix: "Remove '#include' or switch the language toolbar to C++.",
        };
      }
    }

    // Check for delimiter mismatches before Function() evaluation
    if (openParens > 0) {
      return {
        hasError: true,
        type: 'syntax',
        title: "SyntaxError: missing ')' after argument list or expression",
        line: lines.length,
        codeSnippet: lines[lines.length - 1].trim(),
        whereIsMistake: `Line ${lines.length}: Unclosed parenthesis '('`,
        whyIsError: "An opening '(' was never closed with a matching ')'. JavaScript cannot parse an expression with unbalanced parentheses.",
        howToFix: "Add ')' to close the open expression.",
      };
    }

    if (openBrackets > 0) {
      return {
        hasError: true,
        type: 'syntax',
        title: "SyntaxError: missing ']' after array elements",
        line: lines.length,
        codeSnippet: lines[lines.length - 1].trim(),
        whereIsMistake: `Line ${lines.length}: Unclosed bracket '['`,
        whyIsError: "An opening square bracket '[' was never closed with a matching ']'.",
        howToFix: "Add ']' to close the array or index access.",
      };
    }

    if (openBraces > 0) {
      return {
        hasError: true,
        type: 'syntax',
        title: "SyntaxError: missing '}' after function or block body",
        line: lines.length,
        codeSnippet: lines[lines.length - 1].trim(),
        whereIsMistake: `Line ${lines.length}: Unclosed curly brace '{'`,
        whyIsError: "An opening curly brace '{' was never closed with a matching '}'.",
        howToFix: "Add '}' to close the block scope.",
      };
    }

    // Evaluate JavaScript syntax safely
    try {
      const sanitized = code
        .replace(/:\s*(string|number|boolean|any|void|JwtPayload|null)\b/g, '')
        .replace(/interface\s+\w+\s*\{[\s\S]*?\}/g, '');
      new Function(sanitized);
    } catch (err: unknown) {
      const syntaxMsg = err instanceof Error ? err.message : String(err);
      let errLine = 1;

      for (let i = 0; i < lines.length; i++) {
        const l = lines[i].trim();
        if (l.startsWith('print(') || l.includes('import .') || l.includes('const =') || l.includes('let =')) {
          errLine = i + 1;
          break;
        }
      }

      const failingSnippet = lines[errLine - 1] ? lines[errLine - 1].trim() : code.slice(0, 40);

      // Human-friendly translation of V8's cryptic messages
      let readableWhy = `The JavaScript engine's parser cannot understand this grammar: "${syntaxMsg}".`;
      let readableFix = `Check line ${errLine} for mismatched brackets, unclosed quotes, or foreign language syntax.`;

      if (syntaxMsg.includes("Unexpected token '}'")) {
        readableWhy = `The JavaScript engine reached the end of the script while waiting for an unclosed delimiter (like '(', '[', or '{') to be closed.`;
        readableFix = `Ensure all opened brackets, parentheses, and braces are properly closed.`;
      }

      return {
        hasError: true,
        type: 'syntax',
        title: `SyntaxError: ${syntaxMsg}`,
        line: errLine,
        codeSnippet: failingSnippet,
        whereIsMistake: `Line ${errLine}: "${failingSnippet}"`,
        whyIsError: readableWhy,
        howToFix: readableFix,
      };
    }
  }

  // 8. Code is clean!
  return {
    hasError: false,
    type: 'clean',
    title: `Code Syntax Clean & Valid (${language.toUpperCase()})`,
    whereIsMistake: 'No syntax mistakes detected.',
    whyIsError: `Your ${language.toUpperCase()} code grammar is valid and ready to execute.`,
    howToFix: 'Click "Run Code" (or press Ctrl+Enter) to evaluate output in the sandbox.',
  };
}

import fs from 'fs';
import path from 'path';
import { spawn, execSync } from 'child_process';

const WORKSPACES_ROOT = path.join(process.cwd(), '.workspaces', 'repair-jobs');

/**
 * Ensure workspaces root exists
 */
export function ensureWorkspacesRoot(): string {
  if (!fs.existsSync(WORKSPACES_ROOT)) {
    fs.mkdirSync(WORKSPACES_ROOT, { recursive: true });
  }
  return WORKSPACES_ROOT;
}

/**
 * Path Traversal Guard:
 * Resolves a path relative to workspaceDir and asserts that the canonical
 * destination is strictly inside workspaceDir.
 */
export function assertSafePath(workspaceDir: string, relativePath: string): string {
  // Normalize and prevent null bytes or empty path
  if (!relativePath || relativePath.includes('\0')) {
    throw new Error('SECURITY_VIOLATION: Invalid or malformed file path.');
  }

  const cleanWorkspace = path.resolve(workspaceDir);
  const resolvedTarget = path.resolve(cleanWorkspace, relativePath);

  // Strict boundary check: target must be under workspace directory
  if (!resolvedTarget.startsWith(cleanWorkspace + path.sep) && resolvedTarget !== cleanWorkspace) {
    throw new Error(`SECURITY_VIOLATION: Path traversal attempt blocked for "${relativePath}".`);
  }

  return resolvedTarget;
}

/**
 * Redact secrets and sensitive tokens from logs and terminal outputs
 */
export function redactSecrets(text: string): string {
  if (!text) return text;
  let sanitized = text;

  // Mask common secret patterns: bearer tokens, github tokens, secrets
  sanitized = sanitized.replace(/gh[pousr]_[A-Za-z0-9_]{16,255}/g, 'ghp_REDACTED_SECRET');
  sanitized = sanitized.replace(/Bearer\s+[A-Za-z0-9\-._~+/]+=*/gi, 'Bearer REDACTED_TOKEN');
  sanitized = sanitized.replace(/(AUTH_SECRET|CLIENT_SECRET|API_KEY|TOKEN)=["']?[A-Za-z0-9\-_./+=]+["']?/gi, '$1=REDACTED');
  sanitized = sanitized.replace(/GOCSPX-[A-Za-z0-9\-_]+/g, 'GOCSPX-REDACTED');

  return sanitized;
}

/**
 * Command Allowlist Guard:
 * Only permits strictly vetted development and testing commands.
 */
const ALLOWED_EXECUTABLES = new Set(['npm', 'npm.cmd', 'npx', 'npx.cmd', 'node', 'node.exe']);
const FORBIDDEN_OPERATORS = [';', '&&', '||', '|', '`', '$(', '\n', '\r'];
const FORBIDDEN_COMMANDS = [/\brm\b/i, /\brmdir\b/i, /\bdel\b/i, /\bcurl\b/i, /\bwget\b/i, /\bnc\b/i, /\bbash\b/i, /\bpowershell\b/i, /\bcmd(\.exe)?\b/i];

export function validateCommand(command: string): { executable: string; args: string[] } {
  const trimmed = command.trim();
  if (!trimmed) {
    throw new Error('EMPTY_COMMAND: No command provided to sandbox runner.');
  }

  // Reject shell command chaining or injection characters
  for (const op of FORBIDDEN_OPERATORS) {
    if (trimmed.includes(op)) {
      throw new Error(`SECURITY_VIOLATION: Command contains unpermitted shell operator "${op}".`);
    }
  }

  for (const regex of FORBIDDEN_COMMANDS) {
    if (regex.test(trimmed)) {
      throw new Error(`SECURITY_VIOLATION: Command contains unpermitted command pattern "${regex.source}".`);
    }
  }

  const parts = trimmed.split(/\s+/);
  const executable = parts[0];
  const args = parts.slice(1);

  if (!ALLOWED_EXECUTABLES.has(executable.toLowerCase())) {
    throw new Error(`SECURITY_VIOLATION: Executable "${executable}" is not allowlisted for sandbox execution. Allowed: npm, npx, node.`);
  }

  return { executable, args };
}

/**
 * Initialize isolated disposable workspace for a repair job
 */
export async function initializeWorkspace(
  jobId: string,
  initialFiles?: Record<string, string>
): Promise<string> {
  const root = ensureWorkspacesRoot();
  const workspaceDir = path.join(root, jobId);

  if (fs.existsSync(workspaceDir)) {
    fs.rmSync(workspaceDir, { recursive: true, force: true });
  }

  fs.mkdirSync(workspaceDir, { recursive: true });

  if (initialFiles) {
    for (const [relPath, content] of Object.entries(initialFiles)) {
      const targetPath = assertSafePath(workspaceDir, relPath);
      fs.mkdirSync(path.dirname(targetPath), { recursive: true });
      fs.writeFileSync(targetPath, content, 'utf-8');
    }
  }

  return workspaceDir;
}

/**
 * Copy directory contents into isolated workspace
 */
export async function copyToWorkspace(
  sourceDir: string,
  workspaceDir: string,
  excludePatterns: string[] = ['node_modules', '.git', '.next', '.workspaces', '.data']
): Promise<void> {
  if (!fs.existsSync(sourceDir)) {
    throw new Error(`SOURCE_NOT_FOUND: Source directory "${sourceDir}" does not exist.`);
  }

  const entries = fs.readdirSync(sourceDir, { withFileTypes: true });

  for (const entry of entries) {
    if (excludePatterns.includes(entry.name)) continue;

    const srcPath = path.join(sourceDir, entry.name);
    const destPath = path.join(workspaceDir, entry.name);

    if (entry.isDirectory()) {
      fs.mkdirSync(destPath, { recursive: true });
      await copyToWorkspace(srcPath, destPath, excludePatterns);
    } else if (entry.isFile()) {
      fs.copyFileSync(srcPath, destPath);
    }
  }
}

/**
 * Read file safely from workspace
 */
export function readFile(workspaceDir: string, relativePath: string): string {
  const targetPath = assertSafePath(workspaceDir, relativePath);
  if (!fs.existsSync(targetPath)) {
    throw new Error(`FILE_NOT_FOUND: File "${relativePath}" does not exist in workspace.`);
  }
  return fs.readFileSync(targetPath, 'utf-8');
}

/**
 * Write file safely to workspace
 */
export function writeFile(workspaceDir: string, relativePath: string, content: string): void {
  const targetPath = assertSafePath(workspaceDir, relativePath);
  fs.mkdirSync(path.dirname(targetPath), { recursive: true });
  fs.writeFileSync(targetPath, content, 'utf-8');
}

/**
 * Search workspace code for literal query or symbol
 */
export function searchCode(
  workspaceDir: string,
  query: string,
  maxResults: number = 25
): Array<{ file: string; line: number; text: string }> {
  const results: Array<{ file: string; line: number; text: string }> = [];
  const lowerQuery = query.toLowerCase();

  function scanDir(dir: string) {
    if (results.length >= maxResults) return;
    const entries = fs.readdirSync(dir, { withFileTypes: true });

    for (const entry of entries) {
      if (['node_modules', '.git', '.next', 'dist', 'build'].includes(entry.name)) continue;
      const fullPath = path.join(dir, entry.name);

      if (entry.isDirectory()) {
        scanDir(fullPath);
      } else if (entry.isFile()) {
        try {
          const content = fs.readFileSync(fullPath, 'utf-8');
          const lines = content.split('\n');
          const relPath = path.relative(workspaceDir, fullPath).replace(/\\/g, '/');

          for (let i = 0; i < lines.length; i++) {
            if (lines[i].toLowerCase().includes(lowerQuery)) {
              results.push({
                file: relPath,
                line: i + 1,
                text: lines[i].trim(),
              });
              if (results.length >= maxResults) break;
            }
          }
        } catch {
          // ignore binary or unreadable files
        }
      }
    }
  }

  scanDir(workspaceDir);
  return results;
}

/**
 * Generate standard unified-style diff string between two text snippets
 */
export function createUnifiedDiff(filename: string, beforeText: string, afterText: string): string {
  const beforeLines = beforeText.split('\n');
  const afterLines = afterText.split('\n');

  const diffLines: string[] = [
    `--- a/${filename}`,
    `+++ b/${filename}`,
    `@@ -1,${beforeLines.length} +1,${afterLines.length} @@`,
  ];

  for (const line of beforeLines) {
    if (!afterLines.includes(line)) {
      diffLines.push(`-${line}`);
    } else {
      diffLines.push(` ${line}`);
    }
  }

  for (const line of afterLines) {
    if (!beforeLines.includes(line)) {
      diffLines.push(`+${line}`);
    }
  }

  return diffLines.join('\n');
}

/**
 * Apply validated surgical patch to a file in workspace
 */
export function applyPatch(
  workspaceDir: string,
  patch: {
    file: string;
    beforeCode: string;
    afterCode: string;
  }
): { success: boolean; error?: string; diff: string; additions: number; deletions: number } {
  try {
    const targetPath = assertSafePath(workspaceDir, patch.file);
    if (!fs.existsSync(targetPath)) {
      return {
        success: false,
        error: `Target file "${patch.file}" does not exist in workspace.`,
        diff: '',
        additions: 0,
        deletions: 0,
      };
    }

    const currentContent = fs.readFileSync(targetPath, 'utf-8');

    // Normalizing newlines for reliable matching
    const normCurrent = currentContent.replace(/\r\n/g, '\n');
    const normBefore = patch.beforeCode.replace(/\r\n/g, '\n').trim();
    const normAfter = patch.afterCode.replace(/\r\n/g, '\n').trim();

    if (!normBefore) {
      return {
        success: false,
        error: 'Patch beforeCode cannot be empty.',
        diff: '',
        additions: 0,
        deletions: 0,
      };
    }

    if (!normCurrent.includes(normBefore)) {
      return {
        success: false,
        error: `Anchor mismatch: target lines could not be cleanly located in "${patch.file}".`,
        diff: '',
        additions: 0,
        deletions: 0,
      };
    }

    // Apply substitution cleanly
    const updatedContent = normCurrent.replace(normBefore, normAfter);
    fs.writeFileSync(targetPath, updatedContent, 'utf-8');

    const beforeLines = normBefore.split('\n');
    const afterLines = normAfter.split('\n');
    const additions = Math.max(0, afterLines.length - beforeLines.length) + 1;
    const deletions = Math.max(0, beforeLines.length - afterLines.length) + 1;
    const diff = createUnifiedDiff(patch.file, normBefore, normAfter);

    return {
      success: true,
      diff,
      additions,
      deletions,
    };
  } catch (err: any) {
    return {
      success: false,
      error: err.message || 'Unknown error applying patch.',
      diff: '',
      additions: 0,
      deletions: 0,
    };
  }
}

/**
 * Execute an allowlisted test command inside the sandbox with strict bounds
 */
export async function executeSandboxCommand(
  workspaceDir: string,
  command: string,
  options: {
    timeoutMs?: number;
    maxBufferBytes?: number;
    onOutputChunk?: (chunk: string) => void;
  } = {}
): Promise<{
  exitCode: number;
  stdout: string;
  stderr: string;
  durationMs: number;
  passed: boolean;
  timedOut?: boolean;
}> {
  const timeoutMs = options.timeoutMs ?? 35000;
  const maxBuffer = options.maxBufferBytes ?? 512 * 1024; // 512 KB
  const { executable, args } = validateCommand(command);

  const startTime = Date.now();
  let stdoutAcc = '';
  let stderrAcc = '';
  let timedOut = false;

  // Sanitized environment: explicitly strip sensitive API keys & tokens
  const sanitizedEnv: NodeJS.ProcessEnv = {
    ...process.env,
    NODE_ENV: 'test',
    CI: 'true',
    AUTH_SECRET: undefined,
    GOOGLE_CLIENT_SECRET: undefined,
    GITHUB_CLIENT_SECRET: undefined,
    OPENAI_API_KEY: undefined,
    ANTHROPIC_API_KEY: undefined,
    GEMINI_API_KEY: undefined,
  };

  return new Promise((resolve) => {
    let proc: any = null;
    let timer: NodeJS.Timeout | null = null;
    const isWin = process.platform === 'win32';
    const isNpmScript = isWin && (executable.toLowerCase().startsWith('npm') || executable.toLowerCase().startsWith('npx'));

    try {
      proc = spawn(executable, args, {
        cwd: workspaceDir,
        env: sanitizedEnv,
        shell: isNpmScript, // Only invoke shell for npm/npx batch runners on Windows; run node binary directly
        stdio: ['ignore', 'pipe', 'pipe'],
      });
    } catch (spawnErr: any) {
      return resolve({
        exitCode: 1,
        stdout: '',
        stderr: redactSecrets(spawnErr.message || 'Failed to spawn process'),
        durationMs: Date.now() - startTime,
        passed: false,
      });
    }

    timer = setTimeout(() => {
      timedOut = true;
      try {
        if (proc && !proc.killed) {
          if (isWin) {
            try {
              execSync(`taskkill /pid ${proc.pid} /T /F`);
            } catch {
              proc.kill('SIGKILL');
            }
          } else {
            proc.kill('SIGTERM');
            setTimeout(() => {
              if (proc && !proc.killed) proc.kill('SIGKILL');
            }, 1500);
          }
        }
      } catch {
        // ignore kill errors
      }
    }, timeoutMs);

    proc.stdout?.on('data', (data: Buffer) => {
      const chunk = data.toString('utf-8');
      if (stdoutAcc.length < maxBuffer) {
        stdoutAcc += chunk;
      }
      if (options.onOutputChunk) {
        options.onOutputChunk(redactSecrets(chunk));
      }
    });

    proc.stderr?.on('data', (data: Buffer) => {
      const chunk = data.toString('utf-8');
      if (stderrAcc.length < maxBuffer) {
        stderrAcc += chunk;
      }
      if (options.onOutputChunk) {
        options.onOutputChunk(redactSecrets(chunk));
      }
    });

    proc.on('close', (code: number | null) => {
      if (timer) clearTimeout(timer);
      const durationMs = Date.now() - startTime;
      const exitCode = timedOut ? 124 : (code ?? 0);
      const passed = exitCode === 0;

      if (timedOut) {
        stderrAcc += `\n[SANDBOX TIMEOUT]: Execution exceeded bounded limit of ${timeoutMs}ms and was terminated.`;
      }

      resolve({
        exitCode,
        stdout: redactSecrets(stdoutAcc),
        stderr: redactSecrets(stderrAcc),
        durationMs,
        passed,
        timedOut,
      });
    });

    proc.on('error', (err: any) => {
      if (timer) clearTimeout(timer);
      resolve({
        exitCode: 1,
        stdout: redactSecrets(stdoutAcc),
        stderr: redactSecrets(err.message || 'Subprocess error'),
        durationMs: Date.now() - startTime,
        passed: false,
      });
    });
  });
}

/**
 * Clean up isolated workspace safely
 */
export async function cleanupWorkspace(workspaceDir: string): Promise<void> {
  try {
    const cleanWorkspace = path.resolve(workspaceDir);
    const root = path.resolve(ensureWorkspacesRoot());

    // Never delete outside WORKSPACES_ROOT
    if (cleanWorkspace.startsWith(root + path.sep)) {
      if (fs.existsSync(cleanWorkspace)) {
        fs.rmSync(cleanWorkspace, { recursive: true, force: true });
      }
    }
  } catch (err) {
    console.warn('Sandbox workspace cleanup warning:', err);
  }
}

export const CodePracticeView: React.FC<CodePracticeViewProps> = ({
  onNavigate,
  initialChallengeId,
  onOpenLesson,
}) => {
  // Challenge Selection
  const [selectedChallengeId, setSelectedChallengeId] = useState<string>(
    initialChallengeId || 'type-error-null'
  );

  const activeChallenge =
    PRACTICE_CHALLENGES.find((c) => c.id === selectedChallengeId) || PRACTICE_CHALLENGES[0];

  // Code Editor State
  const [code, setCode] = useState<string>(activeChallenge.initialCode);
  const [language, setLanguage] = useState<SupportedLanguage>(activeChallenge.language);
  const [copied, setCopied] = useState<boolean>(false);
  const [fontSize, setFontSize] = useState<number>(13);
  const [editorTheme, setEditorTheme] = useState<'dark' | 'light'>('dark');

  // Execution & Output State
  const [isRunning, setIsRunning] = useState<boolean>(false);
  const [terminalOutput, setTerminalOutput] = useState<string[]>([]);
  const [executionTimeMs, setExecutionTimeMs] = useState<number | null>(null);
  const [executionStatus, setExecutionStatus] = useState<'idle' | 'success' | 'error'>('idle');

  // Test Results State
  const [testResults, setTestResults] = useState<{ name: string; passed: boolean; message: string }[]>([]);

  // AI Error Explainer State
  const [currentErrorExplanation, setCurrentErrorExplanation] = useState<ErrorExplanation | null>(null);
  const [activeExplainerTab, setActiveExplainerTab] = useState<ExplainerTab>('why');
  const [revealedHints, setRevealedHints] = useState<number>(0);

  // Real-Time Live As-You-Type Diagnostic Sentinel State
  const [liveDiagnostic, setLiveDiagnostic] = useState<LiveDiagnostic | null>(null);

  // Ask AI Tutor Chat State
  const [tutorChatMessages, setTutorChatMessages] = useState<{ role: 'user' | 'tutor'; text: string }[]>(
    []
  );
  const [tutorInput, setTutorInput] = useState<string>('');
  const [isTutorThinking, setIsTutorThinking] = useState<boolean>(false);

  // Real-Time As-You-Type Code Analysis Effect
  useEffect(() => {
    const diag = analyzeCodeLive(code, language, activeChallenge.id);
    setLiveDiagnostic(diag);

    // If an error is detected while typing, update the AI Error Concept Explainer in real-time!
    if (diag.hasError) {
      const explanation = explainError(diag.title, code, language, activeChallenge.id);
      setCurrentErrorExplanation(explanation);
    } else {
      if (executionStatus !== 'error') {
        setCurrentErrorExplanation(null);
      }
    }
  }, [code, language, activeChallenge.id, executionStatus]);

  // Sync when challenge changes
  useEffect(() => {
    setCode(activeChallenge.initialCode);
    setLanguage(activeChallenge.language);
    setTerminalOutput([
      `// Welcome to RepoPilot Code Lab: ${activeChallenge.title}`,
      `// Language: ${activeChallenge.language.toUpperCase()} · Compiler: ${LANGUAGE_OPTIONS.find((l) => l.id === activeChallenge.language)?.compiler}`,
      `// Real-time as-you-type error sentinel & AI concept tutor ACTIVE.`,
      `// Click "Run Code" (or press Ctrl+Enter) to evaluate execution.`,
    ]);
    setExecutionStatus('idle');
    setCurrentErrorExplanation(null);
    setRevealedHints(0);
    setTestResults([]);
    setTutorChatMessages([
      {
        role: 'tutor',
        text: `Hello! I am your AI Error Concept Tutor. As you practice "${activeChallenge.title}" in ${activeChallenge.language.toUpperCase()}, if any syntax or runtime exception occurs, I'll explain the language concept, visualize memory states, and help you fix it!`,
      },
    ]);
  }, [selectedChallengeId]);

  const handleLanguageChange = (newLang: SupportedLanguage) => {
    setLanguage(newLang);
    // Find first challenge matching this language
    const matchingChallenge = PRACTICE_CHALLENGES.find((c) => c.language === newLang);
    if (matchingChallenge && matchingChallenge.id !== activeChallenge.id) {
      setSelectedChallengeId(matchingChallenge.id);
    } else {
      setCode(LANGUAGE_TEMPLATES[newLang]);
      setExecutionStatus('idle');
      setTerminalOutput([
        `// Switched to ${newLang.toUpperCase()} environment (${LANGUAGE_OPTIONS.find((l) => l.id === newLang)?.filename}).`,
        `// Active runtime: ${LANGUAGE_OPTIONS.find((l) => l.id === newLang)?.compiler}.`,
        `// Real-time syntax scanning & AI Error Tutor active for ${newLang.toUpperCase()}.`,
      ]);
    }
  };

  // Execute Code safely across all supported languages
  const runCode = async () => {
    setIsRunning(true);
    setTerminalOutput([`Executing ${language.toUpperCase()} sandbox...`]);
    const start = performance.now();

    await new Promise((resolve) => setTimeout(resolve, 200));

    try {
      if (language === 'python') {
        runPythonCode(code, start);
      } else if (language === 'java') {
        runJavaCode(code, start);
      } else if (language === 'cpp') {
        runCppCode(code, start);
      } else if (language === 'c') {
        runCCode(code, start);
      } else if (language === 'go') {
        runGoCode(code, start);
      } else if (language === 'rust') {
        runRustCode(code, start);
      } else {
        runJavaScriptCode(code, start);
      }
    } catch (err: unknown) {
      handleCaughtError(err, start);
    } finally {
      setIsRunning(false);
    }
  };

  const runJavaScriptCode = (userCode: string, startTime: number) => {
    const logs: string[] = [];

    const customConsole = {
      log: (...args: unknown[]) => {
        logs.push(args.map((a) => (typeof a === 'object' ? JSON.stringify(a, null, 2) : String(a))).join(' '));
      },
      warn: (...args: unknown[]) => {
        logs.push(`⚠️ [WARN] ` + args.map((a) => String(a)).join(' '));
      },
      error: (...args: unknown[]) => {
        logs.push(`💥 [ERROR] ` + args.map((a) => String(a)).join(' '));
      },
    };

    try {
      if (
        activeChallenge.id === 'jwt-exp-flaw' &&
        userCode.includes('if (decodedPayload) {') &&
        !userCode.includes('decodedPayload.exp >')
      ) {
        throw new Error('SecurityVulnerability: Expired token accepted as valid authenticated session');
      }

      if (
        activeChallenge.id === 'react-mutation-bug' &&
        userCode.includes('.push(') &&
        !userCode.includes('[...')
      ) {
        throw new Error('StateMutationError: Array.prototype.push directly mutated state reference without producing a new object reference.');
      }

      const sanitizedCode = userCode
        .replace(/:\s*(string|number|boolean|any|void|JwtPayload|null)\b/g, '')
        .replace(/interface\s+\w+\s*\{[\s\S]*?\}/g, '');

      const runFn = new Function('console', sanitizedCode);
      runFn(customConsole);

      const duration = Math.round(performance.now() - startTime);
      setExecutionTimeMs(duration);
      setExecutionStatus('success');

      if (logs.length === 0) {
        logs.push('Execution completed with no console output (return status 0).');
      }

      setTerminalOutput(logs);
      setCurrentErrorExplanation(null);
      runChallengeTests(userCode);

      try {
        confetti({ particleCount: 60, spread: 70, origin: { y: 0.8 } });
      } catch {
        // ignore
      }
    } catch (err: unknown) {
      handleCaughtError(err, startTime);
    }
  };

  const runPythonCode = (pythonCode: string, startTime: number) => {
    const logs: string[] = [];
    const lines = pythonCode.split('\n');

    let openParens = 0;
    for (let i = 0; i < lines.length; i++) {
      const line = lines[i].trim();
      for (const ch of line) {
        if (ch === '(') openParens++;
        if (ch === ')') openParens--;
      }
      if (line.startsWith('print(') && !line.includes(')')) {
        throw new Error(`SyntaxError: unexpected EOF while parsing - unclosed '(' on line ${i + 1}`);
      }
    }
    if (openParens > 0) {
      throw new Error("SyntaxError: unexpected EOF while parsing (unclosed parenthesis ')')");
    }

    if (
      activeChallenge.id === 'py-index-error' &&
      (pythonCode.includes('range(len(log_entries) + 1)') || pythonCode.includes('log_entries[3]'))
    ) {
      throw new Error('IndexError: list index out of range at line 6 (entry = log_entries[i])');
    }

    if (
      activeChallenge.id === 'py-key-error' &&
      pythonCode.includes("user_session['role']") &&
      !pythonCode.includes('.get(')
    ) {
      throw new Error("KeyError: 'role' in dictionary user_session");
    }

    if (
      activeChallenge.id === 'py-zero-division' &&
      pythonCode.includes('sum(ratings) / len(ratings)') &&
      !pythonCode.includes('if not ratings') &&
      !pythonCode.includes('len(ratings) == 0')
    ) {
      throw new Error('ZeroDivisionError: division by zero in calculate_average_rating');
    }

    // Python runner simulation: evaluate print(...) calls
    const printRegex = /print\s*\(([\s\S]*?)\)/g;
    let match;
    let count = 0;

    const varEnv: Record<string, unknown> = {};
    for (const l of lines) {
      const lineTrim = l.trim();
      const assignMatch = lineTrim.match(/^([a-zA-Z_]\w*)\s*=\s*(.+)$/);
      if (assignMatch && !lineTrim.startsWith('def ') && !lineTrim.startsWith('if ')) {
        const vName = assignMatch[1];
        const vVal = assignMatch[2];
        try {
          varEnv[vName] = eval(vVal.replace(/\bTrue\b/g, 'true').replace(/\bFalse\b/g, 'false').replace(/\bNone\b/g, 'null'));
        } catch {
          varEnv[vName] = vVal;
        }
      }
    }

    while ((match = printRegex.exec(pythonCode)) !== null) {
      count++;
      const rawArg = match[1].trim();
      if (!rawArg) {
        logs.push('');
      } else if (rawArg.startsWith('f"') || rawArg.startsWith("f'")) {
        let fStr = rawArg.slice(2, -1);
        fStr = fStr.replace(/\{([^}]+)\}/g, (_, expr) => {
          return varEnv[expr.trim()] !== undefined ? String(varEnv[expr.trim()]) : expr;
        });
        logs.push(fStr);
      } else if (varEnv[rawArg] !== undefined) {
        logs.push(typeof varEnv[rawArg] === 'object' ? JSON.stringify(varEnv[rawArg]) : String(varEnv[rawArg]));
      } else {
        try {
          const res = new Function(...Object.keys(varEnv), `return (${rawArg})`)(...Object.values(varEnv));
          logs.push(String(res));
        } catch {
          logs.push(rawArg.replace(/^['"]|['"]$/g, ''));
        }
      }
    }

    if (count === 0) {
      logs.push("Python 3.12: Successfully executed with return code 0.");
    }

    const duration = Math.round(performance.now() - startTime);
    setExecutionTimeMs(duration);
    setExecutionStatus('success');
    setTerminalOutput(logs);
    setCurrentErrorExplanation(null);
    runChallengeTests(pythonCode);
  };

  const runJavaCode = (javaCode: string, startTime: number) => {
    const logs: string[] = [];

    if (!javaCode.includes('class ')) {
      throw new Error("error: class, interface, or enum expected");
    }
    if (javaCode.includes('print(') && !javaCode.includes('System.out.println(')) {
      throw new Error("error: cannot find symbol method print() - In Java, use System.out.println(...)");
    }

    if (
      activeChallenge.id === 'java-null-pointer' &&
      javaCode.includes('user.getEmail().toLowerCase()') &&
      !javaCode.includes('user == null') &&
      !javaCode.includes('user.getEmail() == null')
    ) {
      throw new Error('java.lang.NullPointerException: Cannot invoke "String.toLowerCase()" because the return value of "Main$User.getEmail()" is null');
    }

    if (
      activeChallenge.id === 'java-array-oob' &&
      javaCode.includes('i <= scores.length')
    ) {
      throw new Error('java.lang.ArrayIndexOutOfBoundsException: Index 5 out of bounds for length 5 at Main.calculateTotal(Main.java:7)');
    }

    logs.push('[javac] Compiling Main.java with OpenJDK 21... OK');
    logs.push('[java] Executing Main.main():');

    const printRegex = /System\.out\.println\s*\(([\s\S]*?)\)\s*;/g;
    let match;
    let found = false;
    while ((match = printRegex.exec(javaCode)) !== null) {
      found = true;
      const content = match[1].trim();
      logs.push(content.replace(/^"|"$/g, '').replace(/"\s*\+\s*"/g, ''));
    }

    if (!found) {
      logs.push('Java process completed with return code 0.');
    }

    const duration = Math.round(performance.now() - startTime);
    setExecutionTimeMs(duration);
    setExecutionStatus('success');
    setTerminalOutput(logs);
    setCurrentErrorExplanation(null);
    runChallengeTests(javaCode);
  };

  const runCppCode = (cppCode: string, startTime: number) => {
    const logs: string[] = [];

    if (!cppCode.includes('#include')) {
      throw new Error("error: 'cout' was not declared in this scope (missing #include <iostream>)");
    }

    if (
      activeChallenge.id === 'cpp-vector-range' &&
      cppCode.includes('readings.at(readings.size())')
    ) {
      throw new Error('std::out_of_range: vector::_M_range_check: __n (which is 3) >= this->size() (which is 3)');
    }

    if (
      activeChallenge.id === 'cpp-segfault' &&
      cppCode.includes('&banner')
    ) {
      throw new Error('Segmentation fault (core dumped): dereferencing dangling pointer at address 0x7ffd9b8');
    }

    logs.push('[g++] Compiling main.cpp with GCC 13 (-std=c++20)... OK');
    logs.push('[a.out] Program Output:');

    const coutRegex = /std::cout\s*<<\s*([^;]+);/g;
    let match;
    let found = false;
    while ((match = coutRegex.exec(cppCode)) !== null) {
      found = true;
      const parts = match[1].split('<<').map((p) => p.trim());
      const outputLine = parts
        .filter((p) => !p.includes('endl'))
        .map((p) => p.replace(/^"|"$/g, ''))
        .join('');
      logs.push(outputLine);
    }

    if (!found) {
      logs.push('Program exited with return code 0.');
    }

    const duration = Math.round(performance.now() - startTime);
    setExecutionTimeMs(duration);
    setExecutionStatus('success');
    setTerminalOutput(logs);
    setCurrentErrorExplanation(null);
    runChallengeTests(cppCode);
  };

  const runCCode = (cCode: string, startTime: number) => {
    const logs: string[] = [];

    if (!cCode.includes('#include')) {
      throw new Error("error: implicit declaration of function 'printf' (missing #include <stdio.h>)");
    }

    if (
      activeChallenge.id === 'c-buffer-overflow' &&
      cCode.includes('strcpy(buffer, inputUsername)')
    ) {
      throw new Error('*** stack smashing detected ***: terminated (buffer overflow in strcpy into 16-byte buffer)');
    }

    logs.push('[clang] Compiling main.c with Clang 17 (-Wall -Wextra)... OK');
    logs.push('[./main] Standard Output:');

    const printfRegex = /printf\s*\(([\s\S]*?)\)\s*;/g;
    let match;
    let found = false;
    while ((match = printfRegex.exec(cCode)) !== null) {
      found = true;
      const args = match[1].split(',').map((s) => s.trim());
      let fmt = args[0].replace(/^"|"$/g, '').replace(/\\n/g, '');
      if (args.length > 1) {
        fmt = fmt.replace('%s', args[1].replace(/^"|"$/g, ''));
      }
      logs.push(fmt);
    }

    if (!found) {
      logs.push('Process finished with exit code 0.');
    }

    const duration = Math.round(performance.now() - startTime);
    setExecutionTimeMs(duration);
    setExecutionStatus('success');
    setTerminalOutput(logs);
    setCurrentErrorExplanation(null);
    runChallengeTests(cCode);
  };

  const runGoCode = (goCode: string, startTime: number) => {
    const logs: string[] = [];

    if (!goCode.includes('package main')) {
      throw new Error("expected 'package', found import or statement");
    }

    if (
      activeChallenge.id === 'go-nil-pointer' &&
      goCode.includes('return u.Account.Balance') &&

    const printRegex = /fmt\.Print(ln|f)?\s*\(([\s\S]*?)\)/g;
    let match;
    let found = false;
    while ((match = printRegex.exec(goCode)) !== null) {
      found = true;
      const clean = match[2].replace(/^"|"$/g, '').replace(/\\n/g, '');
      logs.push(clean);
    }

    if (!found) {
      logs.push('Go process completed with exit status 0.');
    }

    const duration = Math.round(performance.now() - startTime);
    setExecutionTimeMs(duration);
    setExecutionStatus('success');
    setTerminalOutput(logs);
    setCurrentErrorExplanation(null);
    runChallengeTests(goCode);

    }

    logs.push('[rustc] Compiling with rustc 1.76.0 (opt-level=3)... OK');
    logs.push('[cargo run] Running `target/release/sandbox`:');

    const printRegex = /println!\s*\(([\s\S]*?)\)\s*;/g;
    let match;
    let found = false;
    while ((match = printRegex.exec(rustCode)) !== null) {
      found = true;
      const clean = match[1].replace(/^"|"$/g, '').replace(/\{.*?\}/g, 'data');
      logs.push(clean);
    }

    if (!found) {
      logs.push('Rust program exited successfully.');
    }

    const duration = Math.round(performance.now() - startTime);
    setExecutionTimeMs(duration);
    setExecutionStatus('success');
    setTerminalOutput(logs);
    setCurrentErrorExplanation(null);
    runChallengeTests(rustCode);

      errorMsg,
      ``,
      `AI Error Explainer activated below. Inspect the concept breakdown and guided fix.`,
    ]);

    const explanation = explainError(errorMsg, code, language, activeChallenge.id);
    setCurrentErrorExplanation(explanation);
    setActiveExplainerTab('why');

        passed = currentCode.includes('?.') || currentCode.includes('user && user.subscription');
      } else if (activeChallenge.id === 'ref-error-tdz') {
        passed = currentCode.indexOf('taxRate') < currentCode.indexOf('subtotal * taxRate');
      } else if (activeChallenge.id === 'range-error-recursion') {
        passed = currentCode.includes('<= 1') || currentCode.includes('<= 0');
      } else if (activeChallenge.id === 'async-unhandled-rejection') {
        passed = currentCode.includes('try') && currentCode.includes('catch');
      } else if (activeChallenge.id === 'py-index-error') {
        passed = currentCode.includes('enumerate') || (!currentCode.includes('+ 1') && currentCode.includes('range(len('));
      } else if (activeChallenge.id === 'py-key-error') {
        passed = currentCode.includes('.get(') || currentCode.includes('in user_session');
      } else if (activeChallenge.id === 'py-zero-division') {
        passed = currentCode.includes('if not ratings') || currentCode.includes('len(ratings) == 0') || currentCode.includes('len(ratings) > 0');
      } else if (activeChallenge.id === 'java-null-pointer') {
        passed = currentCode.includes('user == null') || currentCode.includes('user.getEmail() == null') || currentCode.includes('Optional');
      } else if (activeChallenge.id === 'java-array-oob') {
        passed = currentCode.includes('< scores.length') || currentCode.includes(': scores');
      } else if (activeChallenge.id === 'cpp-vector-range') {
        passed = currentCode.includes('.back()') || currentCode.includes('size() - 1');
      } else if (activeChallenge.id === 'cpp-segfault') {
        passed = !currentCode.includes('&banner') && currentCode.includes('return banner');
      } else if (activeChallenge.id === 'c-buffer-overflow') {
        passed = currentCode.includes('snprintf') || currentCode.includes('strncpy');
      } else if (activeChallenge.id === 'go-nil-pointer') {
        passed = currentCode.includes('u.Account == nil') || currentCode.includes('u == nil');

      } else if (activeChallenge.id === 'jwt-exp-flaw') {
        passed = currentCode.includes('.exp >') || currentCode.includes('.exp * 1000');

      return {
        name: tc.name,
        passed,
        message: passed ? `Test passed: Expected "${tc.expectedOutput}"` : `Test failed: Exception thrown`,

                <div className="mt-1 text-[11px] text-emerald-800 font-medium pl-6">

                </div>
              )}
            </div>
          )}

          {/* Execution Terminal */}
          <div className="bg-slate-900 rounded-2xl border border-slate-800 overflow-hidden shadow-sm">
            <div className="px-4 py-2.5 bg-slate-950/80 border-b border-slate-800 flex items-center justify-between text-xs">
              <div className="flex items-center gap-2">
                <TerminalIcon className="w-3.5 h-3.5 text-slate-400" />
                <span className="font-semibold text-slate-300">Terminal Output ({language.toUpperCase()})</span>
              </div>

              <div className="flex items-center gap-3">
                {executionTimeMs !== null && (
                  <span className="text-[10px] text-slate-500 font-mono">
                    Elapsed: {executionTimeMs}ms
                  </span>
                )}
                {executionStatus === 'success' && (
                  <span className="inline-flex items-center gap-1 text-[10px] font-bold text-emerald-400 bg-emerald-950/60 px-2 py-0.5 rounded border border-emerald-800">
                    <CheckCircle2 className="w-3 h-3" />
                    Passed
                  </span>
                )}
                {executionStatus === 'error' && (
                  <span className="inline-flex items-center gap-1 text-[10px] font-bold text-rose-400 bg-rose-950/60 px-2 py-0.5 rounded border border-rose-800">
                    <AlertTriangle className="w-3 h-3" />
                    Exception
                  </span>
                )}
              </div>
            </div>

            <div className="p-4 font-mono text-xs text-slate-300 max-h-48 overflow-y-auto space-y-1 bg-slate-900">
              {terminalOutput.map((line, idx) => (
                <div
                  key={idx}
                  className={
                    line.startsWith('💥')
                      ? 'text-rose-400 font-bold'
                      : line.startsWith('⚠️')
                        ? 'text-amber-400'
                        : line.startsWith('//')
                          ? 'text-slate-500'
                          : line.startsWith('[javac]') || line.startsWith('[g++]') || line.startsWith('[clang]')
                            ? 'text-cyan-400 font-semibold'
                            : 'text-slate-300'
                  }
                >
                  {line}
                </div>
              ))}
            </div>
          </div>

          {/* Test Cases Results */}
          {testResults.length > 0 && (
            <div className="bg-white rounded-2xl border border-slate-200 p-4 space-y-2 shadow-xs">
              <div className="flex items-center justify-between text-xs font-bold text-slate-800">

                {testResults.map((tr, i) => (
                  <div
                    key={i}
                    className={`flex items-center justify-between p-2.5 rounded-xl text-xs ${
                      tr.passed
                        ? 'bg-emerald-50 text-emerald-900 border border-emerald-200'
                        : 'bg-rose-50 text-rose-900 border border-rose-200'
                    }`}
                  >
                    <div className="flex items-center gap-2">
                      {tr.passed ? (
                        <CheckCircle2 className="w-4 h-4 text-emerald-600" />

                        <AlertTriangle className="w-4 h-4 text-rose-600" />

                <div className="py-12 px-4 text-center space-y-3">

                    <CheckCircle2 className="w-6 h-6" />
                  </div>
                  <div>
                    <h4 className="text-xs font-bold text-slate-900 uppercase tracking-wider">

                    </p>
                  </div>

            </div>

    </div>
  );
};

import React, { useState, useEffect } from 'react';
import {
  BookOpen,

  GraduationCap,

import { COURSES } from '@/lib/courses-data';
import { Course, Lesson, NavTab } from '@/lib/types';

interface AcademyViewProps {
  onNavigate: (tab: NavTab) => void;
  onOpenPracticeChallenge?: (challengeId: string) => void;
}

export const AcademyView: React.FC<AcademyViewProps> = ({
  onNavigate,
  onOpenPracticeChallenge,
}) => {
  const { user } = useAuth();

  // State
  const [selectedCourseId, setSelectedCourseId] = useState<string | null>(null);
  const [activeLessonId, setActiveLessonId] = useState<string | null>(null);
  const [searchQuery, setSearchQuery] = useState<string>('');
  const [selectedCategory, setSelectedCategory] = useState<string>('All');
  const [selectedLevel, setSelectedLevel] = useState<string>('All');

  // Progress state persisted in localStorage
  const [completedLessons, setCompletedLessons] = useState<Record<string, boolean>>({});
  const [userXp, setUserXp] = useState<number>(320);
  const [streakDays, setStreakDays] = useState<number>(4);

  // Quiz state for active lesson
  const [quizAnswers, setQuizAnswers] = useState<Record<string, number>>({});
  const [quizSubmitted, setQuizSubmitted] = useState<Record<string, boolean>>({});

  // Certificate Modal state
  const [isCertificateOpen, setIsCertificateOpen] = useState<boolean>(false);

  // Load progress from localStorage
  useEffect(() => {
    try {
      const savedCompleted = localStorage.getItem('repopilot_completed_lessons');
      if (savedCompleted) {
        setCompletedLessons(JSON.parse(savedCompleted));
      }
      const savedXp = localStorage.getItem('repopilot_user_xp');
      if (savedXp) {
        setUserXp(parseInt(savedXp, 10));
      }
    } catch {
      // Local storage fallback
    }
  }, []);

  const saveProgress = (newCompleted: Record<string, boolean>, addedXp: number = 0) => {
    setCompletedLessons(newCompleted);
    const newXp = userXp + addedXp;
    setUserXp(newXp);
    try {
      localStorage.setItem('repopilot_completed_lessons', JSON.stringify(newCompleted));
      localStorage.setItem('repopilot_user_xp', String(newXp));
    } catch {
      // ignore
    }
  };

  const selectedCourse = COURSES.find((c) => c.id === selectedCourseId);

  // Get active lesson
  let activeLesson: Lesson | null = null;
  if (selectedCourse) {
    for (const module of selectedCourse.modules) {
      const found = module.lessons.find((l) => l.id === activeLessonId);
      if (found) {
        activeLesson = found;
        break;
      }
    }
    // Default to first lesson if not selected
    if (!activeLesson && selectedCourse.modules[0]?.lessons[0]) {
      activeLesson = selectedCourse.modules[0].lessons[0];
    }
  }

  // Calculate course completion percentage
  const getCourseProgress = (course: Course) => {
    const totalLessons = course.modules.reduce((acc, m) => acc + m.lessons.length, 0);
    if (totalLessons === 0) return 0;
    const completedCount = course.modules.reduce((acc, m) => {
      return acc + m.lessons.filter((l) => completedLessons[l.id]).length;
    }, 0);
    return Math.round((completedCount / totalLessons) * 100);
  };

  const handleStartCourse = (courseId: string) => {
    setSelectedCourseId(courseId);
    const course = COURSES.find((c) => c.id === courseId);
    if (course && course.modules[0]?.lessons[0]) {
      setActiveLessonId(course.modules[0].lessons[0].id);
    }
    window.scrollTo({ top: 0, behavior: 'smooth' });
  };

  const handleCompleteLesson = (lessonId: string, xpReward: number = 50) => {
    if (!completedLessons[lessonId]) {
      const updated = { ...completedLessons, [lessonId]: true };
      saveProgress(updated, xpReward);

      // Trigger confetti celebration
      try {
        confetti({
          particleCount: 50,
          spread: 60,
          origin: { y: 0.8 },
        });
      } catch {
        // ignore
      }

      // Check if course is now 100% complete
      if (selectedCourse) {
        const progress = getCourseProgress(selectedCourse);
        if (progress >= 95) {
          setTimeout(() => {
            setIsCertificateOpen(true);
            try {
              confetti({
                particleCount: 120,
                spread: 90,
                origin: { y: 0.6 },
              });
            } catch {
              // ignore
            }
          }, 600);
        }
      }
    }
  };

  const handleLaunchPractice = (challengeId: string) => {
    if (onOpenPracticeChallenge) {
      onOpenPracticeChallenge(challengeId);
    }
    onNavigate('practice');
  };

  // Filter courses
  const filteredCourses = COURSES.filter((course) => {
    const matchesCategory =
      selectedCategory === 'All' || course.category === selectedCategory;
    const matchesLevel = selectedLevel === 'All' || course.level === selectedLevel;
    const matchesSearch =
      course.title.toLowerCase().includes(searchQuery.toLowerCase()) ||
      course.description.toLowerCase().includes(searchQuery.toLowerCase()) ||
      course.skillsGained.some((s) => s.toLowerCase().includes(searchQuery.toLowerCase()));
    return matchesCategory && matchesLevel && matchesSearch;
  });

  const categories = [
    'All',
    'Debugging & Errors',
    'Security & Auth',
    'Python & Algorithms',
    'React & Architecture',
  ];

  const levels = ['All', 'Beginner', 'Intermediate', 'Advanced'];

  return (
    <div className="space-y-8 max-w-7xl mx-auto pb-20">
